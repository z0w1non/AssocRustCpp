```rs
fn print_vec(v: &Vec<i32>) {
    for i in v {
        println!("{}", i);
    }
}

fn main() {
    let mut v: Vec<i32> = Vec::new();
    v.push(1);
    v.push(2);
    v.push(3);
    print_vec(&v);
}
```

```cpp
#include <iostream>
#include <vector>

void print_vec(const std::vector<int> & v) {
    for (auto i : v) {
        std::cout << i << std::endl;
    }
}

int main() {
    std::vector<int> v;
    v.push_back(1);
    v.push_back(2);
    v.push_back(3);
    print_vec(v);
	return 0;
}
```

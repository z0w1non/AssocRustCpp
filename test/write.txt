```rs
use std::fmt::Write;
use std::error::Error;

fn main() -> Result<(), Box<dyn Error>> {
    let mut ss: String = String::new();

    write!(&mut ss, "foo")?;
    write!(&mut ss, "{}", 42)?;
    println!("{}", ss);

    Ok(())
}
```
